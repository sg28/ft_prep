class TreeNode {
  constructor(val) {
    this.val = val;
    this.left = null;
    this.right = null;
  }
}

function allPaths(root) {
  let result = [];

  function dfs(node, path) {
    if (!node) return;

    path.push(node.val);  

    if (!node.left && !node.right) {
      result.push([...path]);  
    } else {
      dfs(node.left, path);   
      dfs(node.right, path); 
    }

    path.pop(); // Backtrack to explore other paths
  }

  dfs(root, []);
  return result;
}
